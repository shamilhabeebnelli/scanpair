cmake_minimum_required(VERSION 3.16)
project(ScanPair VERSION 0.1.0 LANGUAGES CXX)
if(WIN32)
  message(FATAL_ERROR "ScanPair currently supports Linux/macOS; on Windows use WSL.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(scanpair_core scanpair.cpp)
target_compile_options(scanpair_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(scanpair main.cpp)
target_link_libraries(scanpair PRIVATE scanpair_core)
target_compile_options(scanpair PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(scanpair_tests tests.cpp)
  target_link_libraries(scanpair_tests PRIVATE scanpair_core)
  add_test(NAME core COMMAND scanpair_tests)
  find_package(Python3 COMPONENTS Interpreter)
  find_program(PDFTOPPM_EXECUTABLE pdftoppm)
  if(Python3_FOUND AND PDFTOPPM_EXECUTABLE)
    add_test(NAME pdf_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/integration_test.py $<TARGET_FILE:scanpair>)
  else()
    message(STATUS "PDF integration tests need Python3 and pdftoppm")
  endif()
endif()
install(TARGETS scanpair RUNTIME DESTINATION bin)
